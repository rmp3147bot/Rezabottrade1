def utc_now():
    return datetime.now(timezone.utc)


def format_utc(dt):
    if dt is None:
        return ""
    return dt.astimezone(timezone.utc).strftime("%Y-%m-%d %H:%M:%S UTC")


def parse_utc_datetime(value):
    if isinstance(value, datetime):
        if value.tzinfo is None:
            return value.replace(tzinfo=timezone.utc)
        return value.astimezone(timezone.utc)

    if isinstance(value, str):
        text = value.strip()
        if text.endswith(" UTC"):
            text = text[:-4]

        try:
            return datetime.strptime(text, "%Y-%m-%d %H:%M:%S").replace(tzinfo=timezone.utc)
        except Exception:
            pass

        try:
            parsed = datetime.fromisoformat(text)
            if parsed.tzinfo is None:
                parsed = parsed.replace(tzinfo=timezone.utc)
            return parsed.astimezone(timezone.utc)
        except Exception:
            pass

    return None


def restore_signal_datetime_fields(signal):
    if not isinstance(signal, dict):
        return signal

    source_candle = signal.get("source_candle")
    if isinstance(source_candle, str):
        parsed = parse_utc_datetime(source_candle)
        if parsed is not None:
            signal["source_candle"] = parsed

    signal_time = signal.get("time")
    if isinstance(signal_time, str):
        parsed = parse_utc_datetime(signal_time)
        if parsed is not None:
            signal["time"] = parsed

    return signal


def parse_xt_timestamp(value):
    return datetime.fromtimestamp(int(value) / 1000.0, timezone.utc)


def parse_okx_timestamp(value):
    return datetime.fromtimestamp(int(value) / 1000.0, timezone.utc)

def get_strategy_mode():
    """
    تشخیص حالت معاملاتی بر اساس ساعت فعلی UTC
    Returns: "REVERSE" or "TREND"
    """
    now = utc_now()
    hour = now.hour
    
    # بررسی ساعت‌های REVERSE
    for start, end in REVERSE_HOURS:
        if start <= hour < end:
            return "REVERSE"
    
    # بررسی ساعت‌های TREND
    for start, end in TREND_HOURS:
        if start <= hour < end:
            return "TREND"
    
    # حالت پیش‌فرض (ایمن)
    return "REVERSE"

def is_signal_expired(signal):
    signal_time = signal.get("time")
    if signal_time is None:
        return True
    
    if isinstance(signal_time, str):
        signal_time = parse_utc_datetime(signal_time)
    
    if signal_time is None:
        return True
    
    age = (utc_now() - signal_time).total_seconds()
    return age > SIGNAL_EXPIRY_SECONDS

def atomic_write_json(path, data):
    ensure_data_dir()
    temp = path + ".tmp"
    try:
        with open(temp, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2, default=str)
            f.flush()
            try:
                os.fsync(f.fileno())
            except Exception:
                pass
        os.replace(temp, path)
    except Exception as exc:
        log.error("Could not write JSON state %s: %s", path, exc)
        try:
            if os.path.exists(temp):
                os.remove(temp)
        except Exception:
            pass


def read_json_file(path, default):
    if not os.path.exists(path):
        return default
    try:
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f)
    except Exception as exc:
        log.warning("Could not read JSON state %s: %s", path, exc)
        return default

def detect_signal(candle, session_state):
    """
    تشخیص سیگنال با دو حالت:
    - REVERSE: برعکس کندل شکست (Fake Breakout)
    - TREND: همراه کندل شکست (Trend Following)
    """
    if candle is None:
        return None

    if candle.get("confirmed", 0) != 1:
        return None

    if session_state is None:
        return None

    if not session_state.get("ready", False):
        return None

    if session_state.get("signal_generated", False):
        return None

    hyper = session_state.get("hyper")
    if not isinstance(hyper, dict):
        return None

    hyper_high = hyper.get("high")
    hyper_low = hyper.get("low")

    if hyper_high is None or hyper_low is None:
        return None

    high = float(candle["high"])
    low = float(candle["low"])
    open_price = float(candle["open"])
    close = float(candle["close"])

    # ================================================================
    # تشخیص نوع کندل (صعودی/نزولی)
    # ================================================================
    is_bullish = close > open_price
    is_bearish = close < open_price

    if not is_bullish and not is_bearish:
        is_bearish = True

    # ================================================================
    # تشخیص عبور از Hyper
    # ================================================================
    broke_high = high >= float(hyper_high) and close > float(hyper_high)
    broke_low = low <= float(hyper_low) and close < float(hyper_low)

    if not broke_high and not broke_low:
        return None

    # ================================================================
    # تشخیص حالت معاملاتی بر اساس ساعت
    # ================================================================
    mode = get_strategy_mode()
    
    log.info(f"📊 STRATEGY MODE: {mode} at hour {utc_now().hour:02d}:00 UTC")

    # ================================================================
    # محاسبه R = 2 × (High - Low) با محدودیت 250
    # ================================================================
    candle_range_value = high - low
    r = candle_range_value * 2
    
    if r > MAX_R:
        log.info(f"R capped at {MAX_R} (original: {r:.2f})")
        r = float(MAX_R)

    if r < 5:
        log.info(f"R too small ({r:.2f}), using minimum 5")
        r = 5.0

    # ================================================================
    # تعیین جهت پوزیشن بر اساس حالت
    # ================================================================
    if mode == "REVERSE":
        # برعکس کندل شکست
        if is_bullish:
            direction = "SHORT"
        else:
            direction = "LONG"
    else:  # TREND
        # همراه کندل شکست
        if is_bullish:
            direction = "LONG"
        else:
            direction = "SHORT"

    entry = close

    if direction == "LONG":
        initial_stop = entry - r
        initial_tp = entry + (r * TP_R_MULTIPLE)
    else:
        initial_stop = entry + r
        initial_tp = entry - (r * TP_R_MULTIPLE)

    # ================================================================
    # لاگ اطلاعات
    # ================================================================
    log.info(
        "📊 BREAKOUT DETECTED: mode=%s candle_type=%s break_type=%s R=%.2f direction=%s entry=%.2f",
        mode,
        "BULLISH" if is_bullish else "BEARISH",
        "HIGH_BREAK" if broke_high else "LOW_BREAK",
        r,
        direction,
        entry
    )

    return {
        "direction": direction,
        "entry": entry,
        "candle": candle,
        "hyper_high": float(hyper_high),
        "hyper_low": float(hyper_low),
        "r": r,
        "candle_range": candle_range_value,
        "candle_type": "BULLISH" if is_bullish else "BEARISH",
        "break_type": "HIGH_BREAK" if broke_high else "LOW_BREAK",
        "strategy_mode": mode
    }

def build_reference_signal(strategy_signal, breakout_candle):
    if strategy_signal is None or breakout_candle is None:
        return None

    direction = strategy_signal.get("direction")
    if direction not in ("LONG", "SHORT"):
        return None

    entry = float(strategy_signal["entry"])
    r = float(strategy_signal["r"])
    breakout_time = breakout_candle["time"]

    if direction == "LONG":
        initial_stop = entry - r
        initial_tp = entry + (r * TP_R_MULTIPLE)
    else:
        initial_stop = entry + r
        initial_tp = entry - (r * TP_R_MULTIPLE)

    signal_id = f"REZA-{direction}-{breakout_time.strftime('%Y%m%d%H%M')}"

    structure_data = session_state.get("structure", {})
    if not isinstance(structure_data, dict):
        structure_data = {}

    signal = {
        "signal_id": signal_id,
        "time": utc_now(),
        "source_candle": breakout_time,
        "symbol": DISPLAY_SYMBOL,
        "reference_market": "XT BTCUSDT (DATA + EXECUTION)",
        "structure": structure_data.get("structure", ""),
        "structure_direction": structure_data.get("direction", ""),
        "hyper_high": float(strategy_signal["hyper_high"]),
        "hyper_low": float(strategy_signal["hyper_low"]),
        "direction": direction,
        "reference_entry": entry,
        "initial_stop": initial_stop,
        "initial_risk": r,
        "initial_tp": initial_tp,
        "tp_r_multiple": TP_R_MULTIPLE,
        "entry_type": "BREAKOUT_DUAL_MODE",
        "candle_type": strategy_signal.get("candle_type"),
        "break_type": strategy_signal.get("break_type"),
        "candle_range": strategy_signal.get("candle_range"),
        "strategy_mode": strategy_signal.get("strategy_mode"),
        "structure_score": structure_data.get("score", 0)
    }

    log.info(
        "🔥 SIGNAL CREATED | id=%s | mode=%s | direction=%s | entry=%s | stop=%s | R=%s | TP=%s",
        signal_id,
        signal.get("strategy_mode"),
        direction,
        entry,
        initial_stop,
        r,
        initial_tp
    )

    return signal

def emit_signal(signal):
    global last_signal

    if signal is None:
        return

    if is_signal_expired(signal):
        log.warning(f"⏰ SIGNAL EXPIRED: {signal.get('signal_id')}")
        return

    if session_state is None:
        log.error("SIGNAL BLOCKED: no active session state.")
        return

    if session_state.get("signal_generated", False):
        log.warning("SIGNAL BLOCKED: a signal has already been generated in this session.")
        return

    session_state["signal_generated"] = True
    session_state["signal_id"] = signal.get("signal_id")
    save_current_session_state()

def print_live_status():
    if current_candle is None:
        return

    if session_state and session_state.get("ready", False):
        session_status = "READY"
    else:
        session_status = "WAITING_FOR_SESSION"

    mode = get_strategy_mode()
    structure = ""
    direction = ""
    score = 0.0
    hyper_high = None
    hyper_low = None
    signal_generated = False

    if session_state:
        structure = session_state["structure"].get("structure", "")
        direction = session_state["structure"].get("direction", "")
        score = session_state["structure"].get("score", 0)
        hyper_high = session_state["hyper"].get("high")
        hyper_low = session_state["hyper"].get("low")
        signal_generated = session_state.get("signal_generated", False)

    with platform_lock:
        platform_online = platform_state["online"]
        pending_count = platform_state["pending_count"]

    platform_status = "CONNECTED" if platform_online else "WAITING"

    print(
        f"{utc_now().strftime('%H:%M:%S')} UTC | "
        f"MODE={mode} | "
        f"XT {DISPLAY_SYMBOL} | "
        f"LAST={last_price:.2f} | "
        f"LIVE5M O={current_candle['open']:.2f} "
        f"H={current_candle['high']:.2f} "
        f"L={current_candle['low']:.2f} "
        f"C={current_candle['close']:.2f} | "
        f"CLOSED={len(primary_candles)} | "
        f"SESSION={session_status} | "
        f"STRUCTURE={structure} | "
        f"SCORE={score:.2f} | "
        f"HYPER_H={hyper_high} | "
        f"HYPER_L={hyper_low} | "
        f"SIGNAL_LOCK={signal_generated} | "
        f"PLATFORM={platform_status} | "
        f"PENDING={pending_count}"
    )
