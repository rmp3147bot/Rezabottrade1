def utc_now():
    return datetime.now(timezone.utc)


def format_utc(dt):
    if dt is None:
        return ""
    return dt.astimezone(timezone.utc).strftime("%Y-%m-%d %H:%M:%S UTC")


def parse_utc_datetime(value):
    if isinstance(value, datetime):
        if value.tzinfo is None:
            return value.replace(tzinfo=timezone.utc)
        return value.astimezone(timezone.utc)

    if isinstance(value, str):
        text = value.strip()
        if text.endswith(" UTC"):
            text = text[:-4]

        try:
            return datetime.strptime(text, "%Y-%m-%d %H:%M:%S").replace(tzinfo=timezone.utc)
        except Exception:
            pass

        try:
            parsed = datetime.fromisoformat(text)
            if parsed.tzinfo is None:
                parsed = parsed.replace(tzinfo=timezone.utc)
            return parsed.astimezone(timezone.utc)
        except Exception:
            pass

    return None


def restore_signal_datetime_fields(signal):
    if not isinstance(signal, dict):
        return signal

    source_candle = signal.get("source_candle")
    if isinstance(source_candle, str):
        parsed = parse_utc_datetime(source_candle)
        if parsed is not None:
            signal["source_candle"] = parsed

    signal_time = signal.get("time")
    if isinstance(signal_time, str):
        parsed = parse_utc_datetime(signal_time)
        if parsed is not None:
            signal["time"] = parsed

    return signal


def parse_xt_timestamp(value):
    if value is None:
        return None
    try:
        return datetime.fromtimestamp(int(value) / 1000.0, timezone.utc)
    except (ValueError, TypeError, OverflowError):
        return None

def get_strategy_mode():
    """
    تشخیص حالت معاملاتی بر اساس ساعت فعلی UTC
    Returns: "REVERSE" or "TREND"
    """
    now = utc_now()
    hour = now.hour
    
    # بررسی ساعت‌های REVERSE
    for start, end in REVERSE_HOURS:
        if start <= hour < end:
            return "REVERSE"
    
    # بررسی ساعت‌های TREND
    for start, end in TREND_HOURS:
        if start <= hour < end:
            return "TREND"
    
    # حالت پیش‌فرض (ایمن)
    return "REVERSE"

def is_signal_expired(signal):
    signal_time = signal.get("time")

        return True
    
    if isinstance(signal_time, str):
        signal_time = parse_utc_datetime(signal_time)
    
    if signal_time is None:
        return True
    
    age = (utc_now() - signal_time).total_seconds()

# 🔥 RSI CALCULATION
# ============================================================================

def calculate_rsi(candles, period=RSI_PERIOD):
    """
    محاسبه شاخص RSI از کندل‌ها
    """
    if len(candles) < period + 1:
        return None
    
    closes = [c["close"] for c in candles]
    
    gains = []
    losses = []
    
    for i in range(1, len(closes)):
        change = closes[i] - closes[i-1]
        if change >= 0:
            gains.append(change)
            losses.append(0)
        else:
            gains.append(0)
            losses.append(abs(change))
    
    # استفاده از آخرین period داده
    gains = gains[-period:]
    losses = losses[-period:]
    
    avg_gain = sum(gains) / period if gains else 0
    avg_loss = sum(losses) / period if losses else 0
    
    if avg_loss == 0:
        return 100.0
    
    rs = avg_gain / avg_loss
    rsi = 100 - (100 / (1 + rs))
    
    return rsi

def detect_hammer(candle):
    """
    تشخیص الگوی Hammer (چکش) - سیگنال برگشت صعودی
    شرایط:
    - کندل صعودی (Close > Open)
    - سایه پایین >= 2 × بدنه
    - سایه بالا <= بدنه
    """
    body = abs(candle["close"] - candle["open"])
    if body == 0:
        return False
    
    lower_shadow = min(candle["close"], candle["open"]) - candle["low"]
    upper_shadow = candle["high"] - max(candle["close"], candle["open"])
    
    return (candle["close"] > candle["open"] and 
            lower_shadow >= BODY_SHADOW_RATIO * body and 
            upper_shadow <= body)


def detect_shooting_star(candle):
    """
    تشخیص الگوی Shooting Star (ستاره دنباله‌دار) - سیگنال برگشت نزولی
    شرایط:
    - کندل نزولی (Close < Open)
    - سایه بالا >= 2 × بدنه
    - سایه پایین <= بدنه
    """
    body = abs(candle["close"] - candle["open"])
    if body == 0:
        return False
    
    lower_shadow = min(candle["close"], candle["open"]) - candle["low"]
    upper_shadow = candle["high"] - max(candle["close"], candle["open"])
    
    return (candle["close"] < candle["open"] and 
            upper_shadow >= BODY_SHADOW_RATIO * body and 
            lower_shadow <= body)

def atomic_write_json(path, data):
    ensure_data_dir()
    temp = path + ".tmp"
    try:
        with open(temp, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2, default=str)
            f.flush()
            try:
                os.fsync(f.fileno())
            except Exception:
                pass
        os.replace(temp, path)
    except Exception as exc:
        log.error("Could not write JSON state %s: %s", path, exc)
        try:
            if os.path.exists(temp):
                os.remove(temp)
        except Exception:
            pass


def read_json_file(path, default):
    if not os.path.exists(path):
        return default
    try:
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f)
    except Exception as exc:
        log.warning("Could not read JSON state %s: %s", path, exc)
        return default

def read_platform_queue():
    ensure_platform_queue_file()
    signals = []

    try:
        with open(PLATFORM_QUEUE_FILE, "r", encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                try:
                    signal = json.loads(line)
                    if isinstance(signal, dict):
                        signal = restore_signal_datetime_fields(signal)
                        if signal.get("signal_id"):
                            signals.append(signal)
                except Exception:
                    log.warning("Invalid platform queue line ignored.")
    except Exception as exc:
        log.error("Could not read platform queue: %s", exc)

    unique = []
    seen = set()
    for signal in signals:
        signal_id = str(signal.get("signal_id"))
        if signal_id in seen:
            continue
        seen.add(signal_id)
        unique.append(signal)
    return unique


def write_platform_queue(signals):
    ensure_platform_queue_file()
    temp = PLATFORM_QUEUE_FILE + ".tmp"
    try:
        with open(temp, "w", encoding="utf-8") as f:
            for signal in signals:
                f.write(json.dumps(signal, ensure_ascii=False, default=str) + "\n")
            f.flush()
            try:
                os.fsync(f.fileno())
            except Exception:
                pass
        os.replace(temp, PLATFORM_QUEUE_FILE)
    except Exception as exc:
        log.error("Could not write platform queue: %s", exc)
        try:
            if os.path.exists(temp):
                os.remove(temp)
        except Exception:
            pass


def refresh_pending_count():
    signals = read_platform_queue()
    with platform_lock:
        platform_state["pending_count"] = len(signals)
    return len(signals)


def queue_platform_signal(signal):
    signals = read_platform_queue()
    signal = restore_signal_datetime_fields(signal)
    signal_id = str(signal.get("signal_id"))

    existing_ids = {str(x.get("signal_id")) for x in signals}
    if signal_id not in existing_ids:
        signals.append(signal)

    write_platform_queue(signals)
    with platform_lock:
        platform_state["pending_count"] = len(signals)


def remove_platform_signal(signal_id):
    signals = read_platform_queue()
    remaining = [x for x in signals if str(x.get("signal_id")) != str(signal_id)]
    write_platform_queue(remaining)
    with platform_lock:
        platform_state["pending_count"] = len(remaining)

        status = str(data.get("status", "ok")).lower()
        if status not in ("ok", "ready", "online", "healthy"):
            raise RuntimeError(f"Platform unhealthy: {status}")

        with platform_lock:
            platform_state["online"] = True
            platform_state["last_check"] = utc_now()
            platform_state["last_success"] = utc_now()
            platform_state["last_error"] = None

        refresh_pending_count()
        return True

    except Exception as exc:
        with platform_lock:
            platform_state["online"] = False
            platform_state["last_check"] = utc_now()
            platform_state["last_error"] = str(exc)
        refresh_pending_count()
        return False

def validate_platform_ack(data, sent_signal_id):
    if not isinstance(data, dict):
        return False

    returned_signal_id = data.get("signal_id")
    if returned_signal_id is None:
        return False

    if str(returned_signal_id) != str(sent_signal_id):
        log.error("INVALID PLATFORM ACK: signal_id mismatch. sent=%s returned=%s", sent_signal_id, returned_signal_id)
        return False

    ack = bool(data.get("ack", False))
    status = str(data.get("status", "")).lower()
    already_received = bool(data.get("already_received", False))
    already_accepted = bool(data.get("already_accepted", False))

    if ack or already_received or already_accepted or status in ("already_received", "already_accepted"):
        return True

    return False


def send_signal_to_platform(signal):
    if not PLATFORM_ENABLED:
        update_signal_status(signal, "WAITING_FOR_PLATFORM", "WAITING_FOR_PLATFORM", "PLATFORM_DISABLED")
        return {"success": False, "status": "PLATFORM_DISABLED"}

    signal = restore_signal_datetime_fields(signal)
    signal_id = str(signal.get("signal_id"))

    # ✅ بررسی انقضای سیگنال قبل از ارسال
    if is_signal_expired(signal):
        log.warning(f"⏰ SIGNAL EXPIRED BEFORE SEND: {signal_id}")
        update_signal_status(
            signal,
            "EXPIRED",
            "EXPIRED",
            "SIGNAL_EXPIRED_BEFORE_SEND"
        )
        return {"success": False, "status": "EXPIRED"}

    payload = build_platform_payload(signal)

    with platform_delivery_lock:
        try:
            response = platform_http.post(
                PLATFORM_BASE_URL + PLATFORM_SIGNAL_PATH,
                json=payload,
                timeout=PLATFORM_TIMEOUT
            )

            if response.status_code not in (200, 201, 202):
                raise RuntimeError(f"HTTP {response.status_code}: {response.text[:1000]}")

            try:
                data = response.json()
            except Exception:
                data = {}

            if not validate_platform_ack(data, signal_id):
                raise RuntimeError(f"Platform response did not contain a valid ACK: {data}")

            with platform_lock:
                platform_state["online"] = True
                platform_state["last_success"] = utc_now()
                platform_state["last_error"] = None
                platform_state["last_ack_signal"] = signal_id

            remove_platform_signal(signal_id)

            update_signal_status(
                signal,
                "READY_FOR_EXECUTION",
                "ACKNOWLEDGED",
                "PLATFORM_ACK",
                response=data
            )

            return {"success": True, "status": data.get("status", "ACKNOWLEDGED"), "response": data}

        except Exception as exc:
            with platform_lock:
                platform_state["online"] = False
                platform_state["last_error"] = str(exc)

            # اگر سیگنال منقضی شده باشد، در صف قرار نگیرد
            if is_signal_expired(signal):
                log.warning(f"⏰ SIGNAL EXPIRED AFTER ERROR: {signal_id}")
                update_signal_status(
                    signal,
                    "EXPIRED",
                    "EXPIRED",
                    "SIGNAL_EXPIRED_AFTER_ERROR"
                )
                return {"success": False, "status": "EXPIRED", "error": str(exc)}

            queue_platform_signal(signal)

            update_signal_status(
                signal,
                "WAITING_FOR_PLATFORM",
                "WAITING_FOR_PLATFORM",
                "DELIVERY_FAILED_PENDING",
                response={"error": str(exc)}
            )

            return {"success": False, "status": "WAITING_FOR_PLATFORM", "error": str(exc)}


def retry_pending_platform_signals():
    if not PLATFORM_ENABLED:
        return

    signals = read_platform_queue()

    with platform_lock:
        platform_state["pending_count"] = len(signals)

    if not signals:
        return

    if not platform_health_check():
        return

    for signal in list(signals):
        signal = restore_signal_datetime_fields(signal)
        signal_id = str(signal.get("signal_id"))

        # بررسی انقضای سیگنال در صف
        if is_signal_expired(signal):
            log.warning(f"⏰ SIGNAL EXPIRED IN QUEUE: {signal_id} - REMOVING")
            
            remove_platform_signal(signal_id)
            
            update_signal_status(
                signal,
                "EXPIRED",
                "EXPIRED",
                "SIGNAL_EXPIRED_IN_QUEUE"
            )
            
            send_webhook_signal(signal)
            continue

        update_signal_status(
            signal,
            "WAITING_FOR_PLATFORM",
            "RETRYING",
            "PLATFORM_RETRY"
        )

        result = send_signal_to_platform(signal)

        if result.get("success"):
            print()
            print("PLATFORM QUEUE SIGNAL DELIVERED")
            print(f"SIGNAL ID : {signal_id}")
            print("STATUS    : ACKNOWLEDGED")
            print()

        else:
            break

        if PLATFORM_RETRY_SECONDS > 0:
            time.sleep(PLATFORM_RETRY_SECONDS)

    refresh_pending_count()


def platform_monitor_loop():
    if not PLATFORM_ENABLED:
        log.warning("Execution Platform connection disabled.")
        return

    while running:
        try:
            online = platform_health_check()
            if online:
                retry_pending_platform_signals()
        except Exception as exc:
            log.exception("Platform monitor error: %s", exc)
        time.sleep(PLATFORM_HEALTH_INTERVAL)


def start_platform_monitor():
    if not PLATFORM_ENABLED:
        return None
    thread = threading.Thread(target=platform_monitor_loop, name="REZA-PLATFORM-V50.7-MONITOR", daemon=True)
    thread.start()
    return thread

def generate_webhook_signature(payload, secret):
    if not secret:
        return None
    return hmac.new(secret.encode("utf-8"), json.dumps(payload, default=str).encode("utf-8"), hashlib.sha256).hexdigest()


def send_webhook_signal(signal):
    if not WEBHOOK_ENABLED:
        return True

    if not WEBHOOK_URL:
        log.warning("WEBHOOK enabled but URL is empty")
        return False

    try:
        signal_copy = restore_signal_datetime_fields(signal.copy())
        for key in ["time", "source_candle"]:
            if key in signal_copy and hasattr(signal_copy[key], "isoformat"):
                signal_copy[key] = signal_copy[key].isoformat()

        payload = {
            "event": "REZA_SIGNAL",
            "version": VERSION,
            "timestamp": format_utc(utc_now()),
            "signal": signal_copy,
            "metadata": {
                "session": {
                    "start": format_utc(session_state["start"]) if session_state else None,
                    "end": format_utc(session_state["end"]) if session_state else None,
                    "structure": session_state.get("structure", {}).get("structure") if session_state else None,
                    "score": session_state.get("structure", {}).get("score", 0) if session_state else 0
                },
                "platform": {
                    "connected": platform_state.get("online", False),
                    "pending_count": platform_state.get("pending_count", 0)
                }
            }
        }

        headers = {
            "Content-Type": "application/json",
            "User-Agent": "REZA-BOT-V50.7-WEBHOOK"
        }

        if WEBHOOK_SECRET:
            signature = generate_webhook_signature(payload, WEBHOOK_SECRET)
            if signature:
                headers["X-Webhook-Signature"] = signature

        response = requests.post(WEBHOOK_URL, json=payload, headers=headers, timeout=WEBHOOK_TIMEOUT)

        if response.status_code >= 400:
            log.error("WEBHOOK failed: HTTP %s - %s", response.status_code, response.text[:500])
            return False

        log.info("WEBHOOK sent successfully: %s", response.status_code)
        append_signal_event(signal, "WEBHOOK_SENT", status=signal.get("status"), platform_status=signal.get("platform_status"), response={"status_code": response.status_code, "text": response.text[:500]})
        return True

    except Exception as exc:
        log.error("WEBHOOK error: %s", exc)
        return False

def floor_to_tick(value, tick):
    v = Decimal(str(value))
    t = Decimal(str(tick))
    if t <= 0:
        return float(v)
    return float((v / t).to_integral_value(rounding=ROUND_DOWN) * t)

def load_xt_instrument():
    global instrument_info

    data = xt_get("/future/market/v1/public/symbol/detail", {"symbol": XT_SYMBOL})
    result = data.get("result", [])

    if not result:
        raise RuntimeError(f"XT instrument not found: {XT_SYMBOL}")

    if isinstance(result, list):
        if not result:
            raise RuntimeError(f"XT instrument not found: {XT_SYMBOL}")
        x = result[0]
    else:
        x = result

    state = x.get("state", 0)
    if state not in (0, "0") and str(state).lower() not in ("live", "trading"):
        raise RuntimeError(f"{XT_SYMBOL} is not LIVE. state={state}")

    tick = float(x.get("pricePrecision", 0.1))
    if tick >= 1:
        tick = 10 ** (-tick)

    lot = float(x.get("quantityPrecision", 0.01))
    if lot >= 1:
        lot = 10 ** (-lot)

    minimum = float(x.get("minQty", 0.01))

    instrument_info = {
        "price_tick": tick,
        "lot_size": lot,
        "min_size": minimum,
        "state": state
    }

    return instrument_info

def candle_body(c):
    return abs(c["close"] - c["open"])


def candle_range(c):
    return c["high"] - c["low"]


def bullish_candle(c):
    return c["close"] > c["open"]


def bearish_candle(c):
    return c["close"] < c["open"]


def candle_body_ratio(c):
    total_range = candle_range(c)
    if total_range <= 0:
        return 0
    return candle_body(c) / total_range


def calculate_sma(values, period):
    if len(values) < period:
        return None
    return sum(values[-period:]) / period

def find_swing_highs(candles, lookback=2):
    swings = []
    for i in range(lookback, len(candles) - lookback):
        current = candles[i]
        left_highs = [candles[j]["high"] for j in range(i - lookback, i)]
        right_highs = [candles[j]["high"] for j in range(i + 1, i + lookback + 1)]
        if current["high"] >= max(left_highs) and current["high"] >= max(right_highs):
            swings.append((i, current["high"]))
    return swings


def find_swing_lows(candles, lookback=2):
    swings = []
    for i in range(lookback, len(candles) - lookback):
        current = candles[i]
        left_lows = [candles[j]["low"] for j in range(i - lookback, i)]
        right_lows = [candles[j]["low"] for j in range(i + 1, i + lookback + 1)]
        if current["low"] <= min(left_lows) and current["low"] <= min(right_lows):
            swings.append((i, current["low"]))
    return swings


# ============================================================================
# LEG DETECTION
# ============================================================================

def detect_leg(candles):
    if len(candles) < 6:
        return None

    last_three = candles[-3:]
    preceding_three = candles[-6:-3]

    for c in last_three:
        if candle_body_ratio(c) < 0.3:
            return None

    if all(bullish_candle(c) for c in last_three):
        bearish_count = sum(1 for c in preceding_three if bearish_candle(c))
        if bearish_count >= 2:
            return {
                "type": "LEG",
                "direction": "LONG",
                "strength": min(1.0, sum(candle_body(c) for c in last_three) / (max(c["high"] for c in last_three) - min(c["low"] for c in last_three))),
                "candles": last_three
            }

    if all(bearish_candle(c) for c in last_three):
        bullish_count = sum(1 for c in preceding_three if bullish_candle(c))
        if bullish_count >= 2:
            return {
                "type": "LEG",
                "direction": "SHORT",
                "strength": min(1.0, sum(candle_body(c) for c in last_three) / (max(c["high"] for c in last_three) - min(c["low"] for c in last_three))),
                "candles": last_three
            }

    return None


# ============================================================================
# TREND DETECTION
# ============================================================================

def detect_trend(candles):
    if len(candles) < 10:
        return None

    closes = [c["close"] for c in candles]
    sma_5 = calculate_sma(closes, 5)
    sma_10 = calculate_sma(closes, 10)

    if sma_5 is None or sma_10 is None:
        return None

    if sma_5 > sma_10 * 1.001:
        direction = "LONG"
    elif sma_5 < sma_10 * 0.999:
        direction = "SHORT"
    else:
        return None

    start_idx = 0
    for i in range(len(candles) - 10, 0, -1):
        test_sma5 = calculate_sma(closes[:i+5], 5)
        test_sma10 = calculate_sma(closes[:i+10], 10)
        if test_sma5 and test_sma10:
            if direction == "LONG" and test_sma5 <= test_sma10 * 1.001:
                start_idx = i
                break
            elif direction == "SHORT" and test_sma5 >= test_sma10 * 0.999:
                start_idx = i
                break

    trend_candles = candles[start_idx:]
    if len(trend_candles) < 5:
        return None

    high = max(c["high"] for c in trend_candles)
    low = min(c["low"] for c in trend_candles)
    range_total = high - low

    if range_total <= 0:
        strength = 0
    else:
        directional_move = abs(trend_candles[-1]["close"] - trend_candles[0]["open"])
        strength = min(1.0, directional_move / range_total)

    return {
        "type": "TREND",
        "direction": direction,
        "strength": strength,
        "candles": trend_candles,
        "start_idx": start_idx
    }


# ============================================================================
# RANGE DETECTION
# ============================================================================

def detect_range(candles):
    if len(candles) < 6:
        return None

    recent = candles[-6:]
    high = max(c["high"] for c in recent)
    low = min(c["low"] for c in recent)
    range_width = (high - low) / low

    if range_width > 0.03:
        return None

    max_consecutive = 0
    current_color = None
    count = 0

    for c in recent:
        color = "BULL" if bullish_candle(c) else "BEAR"
        if color == current_color:
            count += 1
            max_consecutive = max(max_consecutive, count)
        else:
            current_color = color
            count = 1

    if max_consecutive >= 3:
        return None

    start_idx = 0
    for i in range(len(candles) - 6, 0, -1):
        test_range = candles[i:i+6]
        test_high = max(c["high"] for c in test_range)
        test_low = min(c["low"] for c in test_range)
        test_width = (test_high - test_low) / test_low
        if test_width > 0.03:
            start_idx = i + 1
            break

    range_candles = candles[start_idx:]
    if len(range_candles) < 4:
        return None

    final_high = max(c["high"] for c in range_candles)
    final_low = min(c["low"] for c in range_candles)
    final_width = (final_high - final_low) / final_low

    if final_width > 0.03:
        return None

    return {
        "type": "RANGE",
        "direction": "RANGE",
        "strength": 0.0,
        "candles": range_candles,
        "high": final_high,
        "low": final_low
    }

# STRUCTURE SCORING
# ============================================================================

def calculate_leg_score(candles, leg_result):
    if not leg_result:
        return 0.0

    score = 0.0
    body_sizes = [candle_body_ratio(c) for c in leg_result.get("candles", [])]
    if body_sizes:
        avg_body = sum(body_sizes) / len(body_sizes)
        score += min(0.5, avg_body * 0.5)

    if len(candles) >= 8:
        closes = [c["close"] for c in candles]
        sma_5 = calculate_sma(closes, 5)
        sma_10 = calculate_sma(closes, 10)
        if sma_5 and sma_10:
            direction = leg_result.get("direction", "")
            if direction == "LONG" and sma_5 > sma_10:
                score += 0.3
            elif direction == "SHORT" and sma_5 < sma_10:
                score += 0.3

    score += min(0.2, leg_result.get("strength", 0) * 0.2)
    return min(1.0, score)


def calculate_trend_score(candles, trend_result):
    if not trend_result:
        return 0.0

    score = 0.0
    closes = [c["close"] for c in candles]
    sma_5 = calculate_sma(closes, 5)
    sma_10 = calculate_sma(closes, 10)
    if sma_5 and sma_10 and sma_10 > 0:
        divergence = abs(sma_5 - sma_10) / sma_10
        score += min(0.4, divergence * 20)

    trend_candles = trend_result.get("candles", [])
    length_score = min(0.3, len(trend_candles) / 20 * 0.3)
    score += length_score
    score += min(0.3, trend_result.get("strength", 0) * 0.3)

    return min(1.0, score)


def calculate_range_score(candles, range_result):
    if not range_result:
        return 0.0

    score = 0.0
    range_candles = range_result.get("candles", [])
    if range_candles:
        high = max(c["high"] for c in range_candles)
        low = min(c["low"] for c in range_candles)
        if low > 0:
            volatility = (high - low) / low
            score += max(0, 0.5 - volatility * 10)

    length_score = min(0.3, len(range_candles) / 10 * 0.3)
    score += length_score

    if range_candles:
        bulls = sum(1 for c in range_candles if bullish_candle(c))
        bears = len(range_candles) - bulls
        if bulls > 0 and bears > 0:
            ratio = min(bulls, bears) / len(range_candles)
            score += ratio * 0.2

    return min(1.0, score)

# MAIN STRUCTURE DETECTION
# ============================================================================

def calculate_structure(candles):
    if len(candles) != SESSION_CANDLES:
        return None

    if any(c.get("confirmed") != 1 for c in candles):
        return None

    leg_result = detect_leg(candles)
    trend_result = detect_trend(candles)
    range_result = detect_range(candles)

    patterns = []

    if leg_result:
        score = calculate_leg_score(candles, leg_result)
        patterns.append({
            "structure": "LEG",
            "direction": leg_result["direction"],
            "score": score,
            "result": leg_result
        })

    if trend_result:
        score = calculate_trend_score(candles, trend_result)
        patterns.append({
            "structure": "BULLISH" if trend_result["direction"] == "LONG" else "BEARISH",
            "direction": trend_result["direction"],
            "score": score,
            "result": trend_result
        })

    if range_result:
        score = calculate_range_score(candles, range_result)
        patterns.append({
            "structure": "RANGE",
            "direction": "RANGE",
            "score": score,
            "result": range_result
        })

    if not patterns:
        fallback = {
            "structure": "RANGE",
            "direction": "RANGE",
            "score": 0.0,
            "result": None
        }
        patterns.append(fallback)

    best = max(patterns, key=lambda p: p["score"])

    if best["result"]:
        result = best["result"]
        candles_data = result.get("candles", candles)
    else:
        candles_data = candles

    high = max(c["high"] for c in candles_data)
    low = min(c["low"] for c in candles_data)

    return {
        "structure": best["structure"],
        "direction": best["direction"],
        "high": high,
        "low": low,
        "swing_high": high,
        "swing_low": low,
        "strength": best["score"],
        "score": best["score"],
        "analysis_start": candles_data[0]["time"] if candles_data else candles[0]["time"],
        "analysis_end": candles_data[-1]["time"] if candles_data else candles[-1]["time"],
        "structure_candles": len(candles_data)
    }

# STRUCTURE-DERIVED HYPER

def calculate_hyper_from_structure(candles, structure):
    if not candles or not structure:
        return None

    structure_type = structure["structure"]

    if structure_type in ["BULLISH", "LEG"] and structure.get("direction") == "LONG":
        start_time = structure.get("analysis_start")
        selected = candles
        if start_time is not None:
            selected = [c for c in candles if c["time"] >= start_time]
        if not selected:
            selected = candles

        high = max(c["high"] for c in selected)
        low = min(c["low"] for c in selected)

        if high > low:
            return {
                "high": high,
                "low": low,
                "source": f"{structure_type}_LONG_LOCKED"
            }

    if structure_type in ["BEARISH", "LEG"] and structure.get("direction") == "SHORT":
        start_time = structure.get("analysis_start")
        selected = candles
        if start_time is not None:
            selected = [c for c in candles if c["time"] >= start_time]
        if not selected:
            selected = candles

        high = max(c["high"] for c in selected)
        low = min(c["low"] for c in selected)

        if high > low:
            return {
                "high": high,
                "low": low,
                "source": f"{structure_type}_SHORT_LOCKED"
            }

    if structure_type == "RANGE":
        start_time = structure.get("analysis_start")
        end_time = structure.get("analysis_end")
        selected = candles

        if start_time is not None and end_time is not None:
            selected = [c for c in candles if c["time"] >= start_time and c["time"] <= end_time]

        if not selected:
            selected = candles

        high = max(c["high"] for c in selected)
        low = min(c["low"] for c in selected)

        if high > low:
            return {
                "high": high,
                "low": low,
                "source": "RANGE_LOCKED"
            }

    return None

def get_previous_12_closed_candles():

def setup_session_locked_levels(state):
    state["ready"] = False
    state["structure"] = None
    state["hyper"] = None
    state["analysis_candles"] = 0

    analysis = get_previous_12_closed_candles()

        return False

    if len(analysis) != SESSION_CANDLES:
        log.error("SESSION INIT FAILED: expected=%s got=%s", SESSION_CANDLES, len(analysis))
        return False

    for candle in analysis:
        if candle.get("confirmed") != 1:
            log.error("SESSION INIT FAILED: unfinished candle found.")
            return False

    structure = calculate_structure(analysis)
    if not structure:
        log.error("SESSION INIT FAILED: Structure unavailable.")
        return False

    log.info(
        "STRUCTURE DETECTED: type=%s direction=%s score=%.2f candles=%s",
        structure.get("structure"),
        structure.get("direction"),
        structure.get("score", 0),
        structure.get("structure_candles", 0)
    )

    debug_data = {
        "time": format_utc(utc_now()),
        "structure": structure,
        "analysis": [{"time": format_utc(c["time"]), "open": c["open"], "high": c["high"], "low": c["low"], "close": c["close"]} for c in analysis]
    }
    with open(STRUCTURE_DEBUG_FILE, "a", encoding="utf-8") as f:
        f.write(json.dumps(debug_data, default=str) + "\n")

    hyper = calculate_hyper_from_structure(analysis, structure)
    if not hyper:
        log.error("SESSION INIT FAILED: Hyper unavailable | structure=%s", structure)
        return False

    hyper_high = hyper.get("high")
    hyper_low = hyper.get("low")

    if hyper_high is None or hyper_low is None:
        log.error("SESSION INIT FAILED: Hyper High/Low missing.")
        return False

    if hyper_high <= hyper_low:
        log.error("SESSION INIT FAILED: Invalid Hyper range | high=%s low=%s", hyper_high, hyper_low)
        return False

    state["structure"] = structure
    state["hyper"] = hyper
    state["analysis_candles"] = len(analysis)
    state["ready"] = True

    save_session(state)
    save_current_session_state()

    log.info(
        "SESSION READY | start=%s | analysis=%s | structure=%s | direction=%s | score=%.2f | hyper_high=%s | hyper_low=%s | source=%s",
        format_utc(state["start"]),
        len(analysis),
        structure.get("structure"),
        structure.get("direction"),
        structure.get("score", 0),
        hyper_high,
        hyper_low,
        hyper.get("source")
    )

    return True

# 🔥 SIGNAL DETECTION - DUAL MODE + REVERSAL PATTERNS
# ============================================================================

def detect_signal(candle, session_state):
    """
    تشخیص سیگنال با ترکیب:
    1. شکست هایپر (شرط اصلی)
    2. Hammer + RSI<40 → LONG (هم‌جهت با کندل)
    3. Shooting Star + RSI>60 → SHORT (هم‌جهت با کندل)
    4. در غیر این صورت → DUAL MODE (برعکس کندل)
    """
    if candle is None:
        return None

    if candle.get("confirmed", 0) != 1:
        return None

    if session_state is None:
        return None

    if not session_state.get("ready", False):
        return None

    if session_state.get("signal_generated", False):
        return None

    hyper = session_state.get("hyper")
    if not isinstance(hyper, dict):
        return None

    hyper_high = hyper.get("high")
    hyper_low = hyper.get("low")

    if hyper_high is None or hyper_low is None:
        return None

    high = float(candle["high"])
    low = float(candle["low"])
    open_price = float(candle["open"])
    close = float(candle["close"])

    # ================================================================
    # 1️⃣ تشخیص نوع کندل (صعودی/نزولی)
    # ================================================================
    is_bullish = close > open_price
    is_bearish = close < open_price

    if not is_bullish and not is_bearish:
        is_bearish = True

    # ================================================================
    # 2️⃣ تشخیص عبور از Hyper (شرط اصلی شکست)
    # ================================================================
    broke_high = high >= float(hyper_high) and close > float(hyper_high)
    broke_low = low <= float(hyper_low) and close < float(hyper_low)

    # اگر از هایپر عبور نکرده → برگرد
    if not broke_high and not broke_low:
        return None

    # ================================================================
    # 3️⃣ تشخیص حالت معاملاتی بر اساس ساعت
    # ================================================================
    mode = get_strategy_mode()
    log.info(f"📊 STRATEGY MODE: {mode} at hour {utc_now().hour:02d}:00 UTC")

    # ================================================================
    # 4️⃣ محاسبه R = 2 × (High - Low) با محدودیت 250
    # ================================================================
    candle_range_value = high - low
    r = candle_range_value * 2
    
    if r > MAX_R:
        log.info(f"R capped at {MAX_R} (original: {r:.2f})")
        r = float(MAX_R)

    if r < 5:
        log.info(f"R too small ({r:.2f}), using minimum 5")
        r = 5.0

    # ================================================================
    # 5️⃣ محاسبه RSI از ۱۴ کندل آخر
    # ================================================================
    rsi = calculate_rsi(primary_candles, RSI_PERIOD)
    
    # ================================================================
    # 6️⃣ تشخیص الگوهای کندلی
    # ================================================================
    is_hammer = detect_hammer(candle)
    is_shooting_star = detect_shooting_star(candle)

    # ================================================================
    # 7️⃣ منطق اصلی:
    #    اگر الگوی برگشت + RSI تأیید شد → هم‌جهت
    #    در غیر این صورت → DUAL MODE (برعکس)
    # ================================================================

    # ================================================================
    # حالت A: کندل صعودی + Hammer + RSI<40 → LONG (هم‌جهت)
    # ================================================================
    if is_bullish and is_hammer and rsi is not None and rsi < RSI_OVERSOLD:
        log.info(f"🔨 BULLISH HAMMER + RSI={rsi:.2f} < {RSI_OVERSOLD} → LONG SIGNAL")
        
        return {
            "direction": "LONG",
            "entry": close,
            "candle": candle,
            "hyper_high": float(hyper_high),
            "hyper_low": float(hyper_low),
            "r": r,
            "candle_range": candle_range_value,
            "candle_type": "BULLISH",
            "break_type": "HAMMER_REVERSAL",
            "strategy_mode": mode,
            "touch_type": "HAMMER",
            "rsi": rsi,
            "pattern": "HAMMER",
            "is_reversal_confirmed": True
        }

    # ================================================================
    # حالت B: کندل نزولی + Shooting Star + RSI>60 → SHORT (هم‌جهت)
    # ================================================================
    if is_bearish and is_shooting_star and rsi is not None and rsi > RSI_OVERBOUGHT:
        log.info(f"💫 BEARISH SHOOTING STAR + RSI={rsi:.2f} > {RSI_OVERBOUGHT} → SHORT SIGNAL")
        
        return {
            "direction": "SHORT",
            "entry": close,
            "candle": candle,
            "hyper_high": float(hyper_high),
            "hyper_low": float(hyper_low),
            "r": r,
            "candle_range": candle_range_value,
            "candle_type": "BEARISH",
            "break_type": "SHOOTING_STAR_REVERSAL",
            "strategy_mode": mode,
            "touch_type": "SHOOTING_STAR",
            "rsi": rsi,
            "pattern": "SHOOTING_STAR",
            "is_reversal_confirmed": True
        }

    # ================================================================
    # حالت C: شروط تأیید نشد → DUAL MODE (برعکس کندل شکست)
    # ================================================================
    log.info(f"📊 No reversal confirmation → Using DUAL MODE (OPPOSITE)")

    # DUAL MODE: برعکس کندل شکست
    if is_bullish:
        direction = "SHORT"   # کندل صعودی → پوزیشن شورت (برعکس)
    else:
        direction = "LONG"    # کندل نزولی → پوزیشن لانگ (برعکس)

    entry = close

    return {
        "direction": direction,
        "entry": entry,
        "candle": candle,
        "hyper_high": float(hyper_high),
        "hyper_low": float(hyper_low),
        "r": r,
        "candle_range": candle_range_value,
        "candle_type": "BULLISH" if is_bullish else "BEARISH",
        "break_type": "HIGH_BREAK" if broke_high else "LOW_BREAK",
        "strategy_mode": mode,
        "touch_type": "DUAL_MODE_FAKE_BREAKOUT",
        "rsi": rsi,
        "pattern": "DUAL_MODE",
        "is_reversal_confirmed": False
    }

# BUILD REFERENCE SIGNAL
# ============================================================================

def build_reference_signal(strategy_signal, breakout_candle):
    if strategy_signal is None or breakout_candle is None:
        return None

    direction = strategy_signal.get("direction")
    if direction not in ("LONG", "SHORT"):
        return None

    entry = float(strategy_signal["entry"])
    r = float(strategy_signal["r"])
    breakout_time = breakout_candle["time"]

    if direction == "LONG":
        initial_stop = entry - r
        initial_tp = entry + (r * TP_R_MULTIPLE)
    else:
        initial_stop = entry + r
        initial_tp = entry - (r * TP_R_MULTIPLE)

    signal_id = f"REZA-{direction}-{breakout_time.strftime('%Y%m%d%H%M')}"

    structure_data = session_state.get("structure", {})
    if not isinstance(structure_data, dict):
        structure_data = {}

    signal = {
        "signal_id": signal_id,
        "time": utc_now(),
        "source_candle": breakout_time,
        "symbol": DISPLAY_SYMBOL,
        "reference_market": "XT BTCUSDT (DATA + EXECUTION)",
        "structure": structure_data.get("structure", ""),
        "structure_direction": structure_data.get("direction", ""),
        "hyper_high": float(strategy_signal["hyper_high"]),
        "hyper_low": float(strategy_signal["hyper_low"]),
        "direction": direction,
        "reference_entry": entry,
        "initial_stop": initial_stop,
        "initial_risk": r,
        "initial_tp": initial_tp,
        "tp_r_multiple": TP_R_MULTIPLE,
        "entry_type": strategy_signal.get("pattern", "BREAKOUT_DUAL_MODE"),
        "candle_type": strategy_signal.get("candle_type", ""),
        "break_type": strategy_signal.get("break_type", ""),
        "candle_range": strategy_signal.get("candle_range", 0),
        "strategy_mode": strategy_signal.get("strategy_mode", ""),
        "structure_score": structure_data.get("score", 0),
        "touch_type": strategy_signal.get("touch_type", ""),
        "rsi": strategy_signal.get("rsi", 0),
        "pattern": strategy_signal.get("pattern", ""),
        "is_reversal_confirmed": strategy_signal.get("is_reversal_confirmed", False)
    }

    log.info(
        "🔥 SIGNAL CREATED | id=%s | mode=%s | direction=%s | entry=%s | stop=%s | R=%s | TP=%s | pattern=%s | RSI=%.2f | reversal=%s",
        signal_id,
        signal.get("strategy_mode"),
        direction,
        entry,
        initial_stop,
        r,
        initial_tp,
        signal.get("pattern"),
        signal.get("rsi", 0),
        signal.get("is_reversal_confirmed")
    )

    return signal

def emit_signal(signal):
    global last_signal

    if signal is None:
        return

    if is_signal_expired(signal):
        log.warning(f"⏰ SIGNAL EXPIRED: {signal.get('signal_id')}")
        return

    if session_state is None:
        log.error("SIGNAL BLOCKED: no active session state.")
        return

    if session_state.get("signal_generated", False):
        log.warning("SIGNAL BLOCKED: a signal has already been generated in this session.")
        return

    session_state["signal_generated"] = True
    session_state["signal_id"] = signal.get("signal_id")
    save_current_session_state()

    last_signal = signal
    save_signal(signal, "WAITING_FOR_PLATFORM")
    queue_platform_signal(signal)
    update_signal_status(signal, "WAITING_FOR_PLATFORM", "WAITING_FOR_PLATFORM", "ADDED_TO_PENDING")

    result = send_signal_to_platform(signal)
    send_webhook_signal(signal)

    mode = signal.get("strategy_mode", "UNKNOWN")
    pattern = signal.get("pattern", "UNKNOWN")
    reversal = signal.get("is_reversal_confirmed", False)

    print()
    print()
    print("=" * 78)
    print(f"🔥 REZA BOT V50.7 — NEW SIGNAL (DUAL MODE + REVERSAL)")
    print("=" * 78)
    print(f"STRATEGY MODE    : {mode}")
    print(f"PATTERN          : {pattern}")
    print(f"REVERSAL CONFIRM : {reversal}")
    print(f"RSI              : {signal.get('rsi', 0):.2f}")
    print(f"SIGNAL ID        : {signal['signal_id']}")
    print(f"TIME             : {format_utc(signal['time'])}")
    print(f"SYMBOL           : {signal['symbol']}")
    print(f"SESSION          : {format_utc(session_state['start'])}")
    print(f"STRUCTURE        : {signal['structure']}")
    print(f"STRUCTURE SCORE  : {signal.get('structure_score', 0):.2f}")
    print(f"HYPER HIGH       : {signal['hyper_high']:.2f}")
    print(f"HYPER LOW        : {signal['hyper_low']:.2f}")
    print(f"DIRECTION        : {signal['direction']}")
    print(f"CANDLE TYPE      : {signal.get('candle_type')}")
    print(f"BREAK TYPE       : {signal.get('break_type')}")
    print(f"R                : {signal['initial_risk']:.2f}")
    print(f"REFERENCE ENTRY  : {signal['reference_entry']:.2f}")
    print(f"INITIAL STOP     : {signal['initial_stop']:.2f}")
    print(f"INITIAL TP       : {signal['initial_tp']:.2f}")
    print(f"ENTRY TYPE       : {signal['entry_type']}")
    print(f"SOURCE CANDLE    : {format_utc(signal['source_candle'])}")
    print("EXECUTION        : SEPARATE PLATFORM (XT)")
    print("SESSION LOCK     : TRUE — NO MORE SIGNALS THIS SESSION")
    print(f"SIGNAL EXPIRY    : {SIGNAL_EXPIRY_SECONDS}s")

    if result.get("success"):
        print("PLATFORM         : CONNECTED ✅")
        print("STATUS           : READY FOR EXECUTION")
    else:
        print("PLATFORM         : NOT AVAILABLE ❌")
        print("STATUS           : WAITING FOR PLATFORM")

    print(f"PENDING COUNT    : {platform_state.get('pending_count', 0)}")
    print("=" * 78)
    print()

def process_live_market(market):
    global current_candle, last_bucket, last_price, market_state, primary_candles, session_state, last_session_start, startup_waiting_for_session

    try:
        price = market["last"]
        now = market["ts"]
        
        if now is None:
            now = utc_now()

        last_price = price
        market_state = market
        market_state["ts"] = now

        save_price(now, market["last"], market.get("mark"), market.get("index"), market.get("funding"))

        bucket = bucket_5m(now)

        if current_candle is None:
            initialize_current_candle()
            return

        if bucket == last_bucket:
            update_candle(current_candle, price)
            return

        # ================================================================
        # بستن کندل قبلی
        # ================================================================
        previous = close_candle(current_candle)
        previous["confirmed"] = 1

        new_closed_candle = False

        if not primary_candles or previous["time"] > primary_candles[-1]["time"]:
            primary_candles.append(previous)
            if len(primary_candles) > MAX_PRIMARY_CANDLES:
                primary_candles = primary_candles[-MAX_PRIMARY_CANDLES:]

            save_closed_candle(previous)
            new_closed_candle = True

            print(
                f"5M CLOSED | {format_utc(previous['time'])} | "
                f"O={previous['open']:.2f} H={previous['high']:.2f} L={previous['low']:.2f} C={previous['close']:.2f} | "
                f"samples={previous.get('samples', 0)} | closed_history={len(primary_candles)}"
            )

        # ================================================================
        # ✅ HH:55 CLOSE — LOCK HYPER FOR NEXT SESSION (FORCE RESET)
        # ================================================================
        is_hh55_candle_close = (
            previous.get("confirmed") == 1
            and previous["time"].minute == 55
            and previous["time"].second == 0
        )

        if is_hh55_candle_close:
            new_session_start = session_start_for(previous["time"] + timedelta(hours=1))
            
            print()
            print("=" * 78)
            print("🔒 HH:55 CLOSED — FORCE LOCKING HYPER FOR NEXT SESSION")
            print(f"NEXT SESSION  : {format_utc(new_session_start)}")
            print(f"HH:55 CLOSED  : {format_utc(previous['time'])}")
            print("HYPER INPUT   : HH:55 + 11 PREVIOUS CLOSED CANDLES")
            print("=" * 78)
            
            rollover_session_if_needed(new_session_start, previous["time"])
            session_created_at_hh55 = True
        else:
            session_created_at_hh55 = False

        # ================================================================
        # تشخیص سیگنال روی کندل بسته شده
        # ================================================================
        if (
            session_state is not None
            and session_state.get("ready", False)
            and new_closed_candle
        ):
            previous_session = session_start_for(previous["time"])
            active_session = session_state.get("start")
            boundary_time = session_state.get("boundary_candle_time")

            candle_belongs_to_active_session = (
                active_session is not None
                and previous_session == active_session
                and (boundary_time is None or previous["time"] >= boundary_time)
            )

            if candle_belongs_to_active_session:
                strategy_signal = detect_signal(previous, session_state)
                if strategy_signal:
                    signal = build_reference_signal(strategy_signal, previous)
                    if signal:
                        emit_signal(signal)

        # ================================================================
        # شروع کندل جدید
        # ================================================================
        current_candle = start_new_candle(bucket, price)
        last_bucket = bucket

        mode = get_strategy_mode()
        print()
        print("NEW LIVE 5M CANDLE (XT PRICES)")
        print(f"BUCKET     : {format_utc(bucket)}")
        print(f"OPEN       : {price:.2f}")
        print("CONFIRM    : 0")
        print(f"MODE       : {mode} ({'LIMIT ENTRY' if mode == 'REVERSE' else 'MARKET ENTRY'})")
        print("PATTERNS   : HAMMER | SHOOTING STAR")
        print("RSI FILTER : LONG<40 | SHORT>60")

        if session_state is not None and session_state.get("ready", False):
            structure_data = session_state.get("structure", {})
            hyper_data = session_state.get("hyper", {})
            print(f"STRUCTURE  : {structure_data.get('structure', 'N/A')}")
            print(f"SCORE      : {structure_data.get('score', 0):.2f}")
            print(f"HYPER_H    : {hyper_data.get('high')}")
            print(f"HYPER_L    : {hyper_data.get('low')}")
            print("STRATEGY   : ACTIVE — CLOSED CANDLE INPUT")
            print("✅ FIRST CANDLE WILL BE CHECKED FOR BREAKOUT")
        else:
            print("STRATEGY   : WAITING FOR SESSION LEVELS")
            print("⏳ Hyper will lock at next HH:55")

        print()

    except Exception as exc:
        log.error(f"🔥 process_live_market ERROR: {exc}")
        print(f"[ERROR] process_live_market: {exc}")
        traceback.print_exc()

def print_live_status():
    """نمایش وضعیت لحظه‌ای"""
    
    if current_candle is None:
        print("⏳ Waiting for initial candle...")
        return

    now = utc_now()
    mode = get_strategy_mode()
    
    if current_candle:
        live_info = format_candle(current_candle)
        samples = current_candle.get("samples", 0)
    else:
        live_info = "NO DATA"
        samples = 0
    
    if session_state and session_state.get("ready", False):
        session_status = "READY"
    else:
        session_status = "WAITING_FOR_SESSION"
    
    hyper_h = None
    hyper_l = None
    if session_state is not None and isinstance(session_state, dict):
        hyper_data = session_state.get("hyper")
        if hyper_data and isinstance(hyper_data, dict):
            hyper_h = hyper_data.get("high")
            hyper_l = hyper_data.get("low")
    
    structure_type = ""
    structure_score = 0.0
    if session_state is not None and isinstance(session_state, dict):
        structure_data = session_state.get("structure")
        if structure_data and isinstance(structure_data, dict):
            structure_type = structure_data.get("structure", "")
            structure_score = structure_data.get("score", 0.0)
    
    signal_generated = session_state.get("signal_generated", False) if session_state else False
    
    with platform_lock:
        platform_online = platform_state["online"]
        pending_count = platform_state["pending_count"]
    
    if not PLATFORM_ENABLED:
        platform_status = "DISABLED"
    else:
        platform_status = "CONNECTED" if platform_online else "WAITING"
    
    status_msg = (
        f"{now.strftime('%H:%M:%S')} UTC | MODE={mode} | XT BTCUSDT | "
        f"LAST={last_price:.2f} | LIVE5M {live_info} | "
        f"SAMPLES={samples} | CLOSED={len(primary_candles)} | "
        f"SESSION={session_status} | STRUCTURE={structure_type} | "
        f"SCORE={structure_score:.2f} | HYPER_H={hyper_h if hyper_h is not None else 'None'} | "
        f"HYPER_L={hyper_l if hyper_l is not None else 'None'} | "
        f"SIGNAL_LOCK={signal_generated} | PLATFORM={platform_status} | "
        f"PENDING={pending_count}"
    )
    
    if os.environ.get("GITHUB_ACTIONS") == "true":
        log.info(status_msg)
    else:
        print(f"\r{status_msg}", end="", flush=True)
